IIT JAM · Mathematics (MA), 2011

  1. Let $a_{n}=\sum_{k=1}^{n} \frac{n}{n^{2}+k}$, for $n \in \mathbb{N}$. Then the sequence $\left\{a_{n}\right\}$ is \begin{enumerate}[label=(\alph*)] \item Convergent \item Diverges to $\infty$ \item Bounded but not convergent \item Neither bounded nor diverges to $\infty$ \end{enumerate}
  2. The number of real roots of the equation $x^{3}+x-1=0$ is \begin{enumerate}[label=(\alph*)] \item 0 \item 1 \item 2 \item 3 \end{enumerate}
  3. The value of $\lim _{n \rightarrow \infty} \sum_{k=1}^{n} \frac{1}{\sqrt{n^{2}+k n}}$ is \begin{enumerate}[label=(\alph*)] \item $2(\sqrt{2}-1)$ \item $2 \sqrt{2}-1$ \item $2-\sqrt{2}$ \item $\frac{1}{2}(\sqrt{2}-1)$ \end{enumerate}
  4. Let $V$ be the region bounded by the planes $x=0, x=2, y=0, z=0$ and $y+z=1$. Then the value of the integral $\iiint_{V} y d x d y d z$ is \begin{enumerate}[label=(\alph*)] \item $\frac{1}{2}$ \item $\frac{4}{3}$ \item $1 $ \item $\frac{1}{3}$ \end{enumerate}
  5. The solution $y(x)$ of the differential equation $\frac{d^{2} y}{d x^{2}}+4 \frac{d y}{d x}+4 y=0$ satisfying the conditions $y(0)=4, \frac{d y}{d x}(0)=8$ is \begin{enumerate}[label=(\alph*)] \item $4 e^{2 x}$ \item $(16 x+4) e^{-2 x}$ \item $4 e^{-2 x}+16 x$ \item $4 e^{-2 x}+16 x e^{2 x}$ \end{enumerate}
  6. If $y^{a}$ is an integrating factor of the differential equation $2 x y d x-\left(3 x^{2}-y^{2}\right) d y=0$, then the value of $a$ is \begin{enumerate}[label=(\alph*)] \item $-4$ \item $4$ \item $-1$ \item $1$ \end{enumerate}
  7. Let $\vec{F}=a y \hat{i}+z \hat{j}+x \hat{k}$ and $C$ be the positively oriented closed curve given by $x^{2}+y^{2}=1, z=0$. If $\oint_{C} \vec{F} \cdot d \vec{r}=\pi$, then the value of $a$ is \begin{enumerate}[label=(\alph*)] \item $-1$ \item $0$ \item $\frac{1}{2}$ \item $1$ \end{enumerate}
  8. Consider the vector field $\vec{F}=(a x+y+a) \hat{i}+\hat{j}-(x+y) \hat{k}$, where $a$ is a constant. If $\vec{F} \cdot \operatorname{cur} 1 \vec{F}=0$, then the value of $a$ is \begin{enumerate}[label=(\alph*)] \item $-1$ \item 0 \item 1 \item $\frac{3}{2}$ \end{enumerate}
  9. Let $G$ denote the group of all $2 \times 2$ invertible matrices with entries from $\mathbb{R}$. Let $$ H_{1}=\{A \in G: \operatorname{det}(A)=1\} \text { and } H_{2}=\{A \in G: A \text { is upper triangular }\} \text {. } $$ Consider the following statements: $P: H_{1}$ is a normal subgroup of $G$ $Q: H_{2}$ is a normal subgroup of $G$. Then \begin{enumerate}[label=(\alph*)] \item Both $P$ and $Q$ are true \item $P$ is true and $Q$ is false \item $P$ is false and $Q$ is true \item Both $P$ and $Q$ are false \end{enumerate}
  10. For $n \in \mathbb{N}$, let $n \mathbb{Z}=\{n k: k \in \mathbb{Z}\}$. Then the number of units of $\mathbb{Z} / 11 \mathbb{Z}$ and $\mathbb{Z} / 12 \mathbb{Z}$, respectively, are \begin{enumerate}[label=(\alph*)] \item 11,12 \item 10,11 \item 10,4 \item 10,8 \end{enumerate}
  11. Let $A$ be a $3 \times 3$ matrix with trace $(A)=3$ and $\operatorname{det}(A)=2$. If 1 is an eigenvalue of $A$, then the eigenvalues of the matrix $A^{2}-2 I$ are \begin{enumerate}[label=(\alph*)] \item $1,2(i-1),-2(i+1)$ \item $-1,2(i-1), 2(i+1)$ \item $1,2(i+1),-2(i+1)$ \item $-1,2(i-1),-2(i+1)$ \end{enumerate}
  12. Let $T: \mathbb{R}^{n} \rightarrow \mathbb{R}^{n}$ be a linear transformation, where $n \geq 2$. For $k \leq n$, let Then $$ E=\left\{v_{1}, v_{2}, \ldots, v_{k}\right\} \subseteq \mathbb{R}^{n} \text { and } F=\left\{T v_{1}, T v_{2}, \ldots, T v_{k}\right\} $$ \begin{enumerate}[label=(\alph*)] \item If $E$ is linearly independent, then $F$ is linearly independent \item If $F$ is linearly independent, then $E$ is linearly independent \item If $E$ is linearly independent, then $F$ is linearly dependent \item If $F$ is linearly independent, then $E$ is linearly dependent \end{enumerate}
  13. For $n \neq m$, let $T_{1}: \mathbb{R}^{n} \rightarrow \mathbb{R}^{m}$ and $T_{2}: \mathbb{R}^{m} \rightarrow \mathbb{R}^{n}$ be linear transformations such that $T_{1} T_{2}$ is bijective. Then \begin{enumerate}[label=(\alph*)] \item $\operatorname{rank}\left(T_{1}\right)=n$ and $\operatorname{rank}\left(T_{2}\right)=m$ \item rank $\left(T_{1}\right)=m$ and $\operatorname{rank}\left(T_{2}\right)=n$ \item $\operatorname{rank}\left(T_{1}\right)=n$ and $\operatorname{rank}\left(T_{2}\right)=n$ \item $\operatorname{rank}\left(T_{1}\right)=m$ and $\operatorname{rank}\left(T_{2}\right)=m$ \end{enumerate}
  14. The set of all $x$ at which the power series $\sum_{n=1}^{\infty} \frac{n}{(2 n+1)^{2}}(x-2)^{3 n}$ converges is \begin{enumerate}[label=(\alph*)] \item $[-1,1)$ \item $[-1,1]$ \item $[1,3)$ \item $[1,3]$ \end{enumerate} % MAINUL
  15. Consider the following subsets of $\mathbb{R}$ : Then $$ E=\left\{\frac{n}{n+1}: n \in \mathbb{N}\right\}, \quad F=\left\{\frac{1}{1-x}: 0 \leq x<1\right\} $$ \begin{enumerate}[label=(\alph*)] \item Both $E$ and $F$ are closed \item $E$ is NOT closed and $F$ is closed \item $E$ is closed and $F$ is NOT closed \item Neither $E$ nor $F$ is closed \end{enumerate}
  16. \begin{enumerate} \item Let $\left\{a_{n}\right\}$ be a sequence of non-negative real numbers such that $\sum_{n=1}^{\infty} a_{n}$ converges, and let $\left\{k_{n}\right\}$ be a strictly increasing sequence of positive integers. Show that $$ \sum_{n=1}^{\infty} a_{k_{n}} \text { also converges. } $$ \item Suppose $f:[0,1] \rightarrow \mathbb{R}$ is differentiable and $f^{\prime}(x) \leq 1$ at every $x \in(0,1)$. If $f(0)=0$ and $f(1)=1$, show that $f(x)=x$ for all $x \in[0,1]$. \end{enumerate}
  17. \begin{enumerate} \item Suppose $f$ is a real valued function defined on an open interval $I$ and differentiable at every $x \in I$. If $[a, b] \subset I$ and $f^{\prime}(a)<0<f^{\prime}(b)$, then show that there exists $c \in(a, b)$ such that $f(c)=\min _{a \leq x \leq b} f(x)$. \item Let $f:(a, b) \rightarrow \mathbb{R}$ be a twice differentiable function such that $f^{\prime \prime}$ is continuous at every point in $(a, b)$. Prove that $$ \lim _{h \rightarrow 0} \frac{f(x+h)-2 f(x)+f(x-h)}{h^{2}}=f^{\prime \prime}(x) $$ for every $x \in(a, b)$. \end{enumerate}
  18. Find all critical points of the following function and check whether the function attains maximum or minimum at each of these points: $$ u(x, y)=x^{4}+y^{4}-2 x^{2}-2 y^{2}+4 x y, \quad(x, y) \in \mathbb{R}^{2} $$
  19. \begin{enumerate} \item Let $\varphi:[a, b] \rightarrow \mathbb{R}$ be differentiable and $[c, d]=\{\varphi(x): a \leq x \leq b\}$, and let $f:[c, d] \rightarrow \mathbb{R}$ be continuous. Let $g:[a, b] \rightarrow \mathbb{R}$ be defined by $g(x)=\int_{c}^{\varphi(x)} f(t) d t$ for $x \in[a, b]$. Then show that $g$ is differentiable and $g^{\prime}(x)=f(\varphi(x)) \varphi^{\prime}(x)$ for all $x \in[a, b]$ \item If $f:[0,1] \rightarrow \mathbb{R}$ is such that $\int_{0}^{\sin x} f(t) d t=\frac{\sqrt{3}}{2} x$ for all $x \in \mathbb{R}$, then find $f\left(\frac{1}{2}\right)$. \end{enumerate}
  20. Find the area of the surface of the solid bounded by the cone $z=3-\sqrt{x^{2}+y^{2}}$ and the paraboloid $z=1+x^{2}+y^{2}$
  21. Obtain the general solution of each of the following differential equations: \begin{enumerate} \item $y-x \frac{d y}{d x}=\frac{d y}{d x} y^{2} e^{y}$. \item $\frac{d y}{d x}=\frac{x+2 y+8}{2 x+y+7}$. \end{enumerate}
  22. \begin{enumerate} \item Determine the values of $b>1$ such that the differential equation $$ x^{2} \frac{d^{2} y}{d x^{2}}+x \frac{d y}{d x}+y=0, \quad 1<x<b $$ satisfying the conditions $y(1)=0=y(b)$ has a nontrivial solution. \item Find $v(x)$ such that $y(x)=e^{4 x} v(x)$ is a particular solution of the differential equation $$ \frac{d^{2} y}{d x^{2}}-8 \frac{d y}{d x}+16 y=\left(2 x+11 x^{10}+21 x^{20}\right) e^{4 x} . $$ \end{enumerate}
  23. \begin{enumerate} \item Change the order of integration in the double integral $$ \int_{-1}^{2}\left(\int_{-x}^{2-x^{2}} f(x, y) d y\right) d x $$ \item Let $\vec{F}=\left(x^{2}-x y^{2}\right) \hat{i}+y^{2} \hat{j}$. Using Green's theorem, evaluate the line integral $\int_{C} \vec{F} \cdot d \vec{r}$, where $C$ is the positively oriented closed curve which is the boundary of the region enclosed by the $x$ - axis and the semi-circle $y=\sqrt{1-x^{2}}$ in the upper half plane. \end{enumerate}
  24. \begin{enumerate} \item If $\vec{F}=\left(x^{2}+y-4\right) \hat{i}+3 x y \hat{j}+\left(2 x z+z^{2}\right) \hat{k}$, then evaluate the surface integral $\iint_{S}(\nabla \times \vec{F}) \cdot \hat{n} d S$, where $S$ is the surface of the cone $z=1-\sqrt{x^{2}+y^{2}}$ lying above the $x y$-plane and $\hat{n}$ is the unit normal to $S$ making an acute angle with $\hat{k}$. \item Show that the series $\sum_{n=1}^{\infty} \frac{x}{\sqrt{n}\left(1+n^{p} x^{2}\right)}$ converges uniformly on $\mathbb{R}$ for $p>1$. \end{enumerate}
  25. \begin{enumerate} \item Find a value of $c$ such that the following system of linear equations has no solution: $$ \begin{gathered} x+2 y+3 z=1, 3 x+7 y+c z=2, 2 x+c y+12 z=3 . \end{gathered} $$ \item Let $V$ be the vector space of all polynomials with real coefficients of degree at most $n$, where $n \geq 2$. Considering elements of $V$ as functions from $\mathbb{R}$ to $\mathbb{R}$, define $$ W=\left\{p \in V: \int_{0}^{1} p(x) d x=0\right\} . $$ Show that $W$ is a subspace of $V$ and $\operatorname{dim}(W)=n$. \end{enumerate}
  26. \begin{enumerate} \item Let $A$ be a $3 \times 3$ real matrix with $\operatorname{det}(A)=6$. Then find $\operatorname{det}(\operatorname{adj} A)$. \item Let $v_{1}$ and $v_{2}$ be non-zero vectors in $\mathbb{R}^{n}, n \geq 3$, such that $v_{2}$ is not a scalar multiple of $v_{1}$. Prove that there exists a linear transformation $T: \mathbb{R}^{n} \rightarrow \mathbb{R}^{n}$ such that $T^{3}=T, T v_{1}=v_{2}$, and $T$ has at least three distinct eigenvalues. \end{enumerate}
  27. \begin{enumerate} \item If $E$ is a subset of $\mathbb{R}$ that does not contain any of its limit points, then prove that $E$ is a countable set. \item Let $f:(a, b) \rightarrow \mathbb{R}$ be a continuous function. If $f$ is uniformly continuous, then prove that there exists a continuous function $g:[a, b] \rightarrow \mathbb{R}$ such that $g(x)=f(x)$ for all $x \in(a, b)$. \end{enumerate}
  28. \begin{enumerate} \item On $\mathbb{R}^{3}$, define a binary operation * as follows: For $(x, y, t),\left(x^{\prime}, y^{\prime}, t^{\prime}\right)$ in $\mathbb{R}^{3}$, $$ (x, y, t)^{*}\left(x^{\prime}, y^{\prime}, t^{\prime}\right)=\left(x+x^{\prime}, y+y^{\prime}, t+t^{\prime}+\frac{1}{2}\left(x^{\prime} y-x y^{\prime}\right)\right) \text {. } $$ Then show that $\left(\mathbb{R}^{3},{ }^{*}\right)$ is a group, and find its center. \item For $k \in \mathbb{N}$, let $k \mathbb{Z}=\{k n: n \in \mathbb{Z}\}$. For any $m, n \in \mathbb{N}$, show that $I=m \mathbb{Z} \cap n \mathbb{Z}$ is an ideal of $\mathbb{Z}$. Further, find the generators of $I$. \end{enumerate}
  29. Let $G$ be a group of order $p^{2}$, where $p$ is a prime number. Let $x \in G$. Prove that $\{y \in G: x y=y x\}=G$.
